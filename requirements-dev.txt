# Verification-only dependencies. Not needed to run or deploy the app.
#
#   python -m pip install -r requirements-dev.txt
#   python -m unittest discover -s tests -t .
#
# pytest is a convenience runner; the suite is plain unittest and also runs
# without it.
#
# openpyxl is what tests/test_real_workbook_interop.py uses to write the input

# openpyxl uninstalled therefore turns the strongest check into a silent skip,
# so install it before signing off a deployment.
